     mcs -langversion:5 -target:exe -main:NapStudioServerV6.NapTestHost -r:System.Windows.Forms.dll \
         -r:System.Drawing.dll -out:NapTest.exe NapServer.cs NapTestHost.cs        (mono-devel)
     javac -d out ../../app/src/main/java/eu/atarihelp/emu10/{NapJson,NapServerKlient,NapApkKlic}.java \
         jvm/TestB321Server.java
     java -cp out TestB321Server NapTest.exe <APK podepsane v2/v3> <pracovni slozka>   -> 28 OK
   (parovani klicem z APK i kodem, podepsane dotazy, stahovani + Range, spatne hodiny v telefonu,
    preposlana appka, blokace, uvolneni kodu, server nedostupny, User-Agent appky v logu serveru)

     javac -d k ../../app/src/main/java/eu/atarihelp/emu10/NapApkKlic.java jvm/NapApkKlicTest.java
     java -cp k NapApkKlicTest s_klicem.apk
